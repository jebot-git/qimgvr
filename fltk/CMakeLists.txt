find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig zlib)
pkg_check_modules(MEDIA REQUIRED IMPORTED_TARGET exiv2 egl gio-2.0)
find_package(OpenCV REQUIRED core imgcodecs imgproc calib3d)
find_package(OpenGL REQUIRED)
add_library(qimgvr-image STATIC image.cpp)
target_compile_features(qimgvr-image PUBLIC cxx_std_17)
target_link_libraries(qimgvr-image PUBLIC fltk::images-shared ${OpenCV_LIBS} PkgConfig::MEDIA)
target_include_directories(qimgvr-image PUBLIC ${OpenCV_INCLUDE_DIRS})
add_library(qimgvr-gltf STATIC gltf-export.cpp)
target_compile_features(qimgvr-gltf PUBLIC cxx_std_17)
target_link_libraries(qimgvr-gltf PUBLIC qimgvr-image)
find_package(Threads REQUIRED)
add_library(qimgvr-model STATIC model/gltf-model.cpp model/gltf-render.cpp model/object-interaction.cpp model/model-loader.cpp model/domed.cpp)
target_compile_features(qimgvr-model PUBLIC cxx_std_17)
target_include_directories(qimgvr-model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/model)
target_link_libraries(qimgvr-model PUBLIC OpenGL::GL Threads::Threads)
add_executable(qimgvr main.cpp canvas.cpp stereo.cpp model-view.cpp)
target_compile_features(qimgvr PRIVATE cxx_std_17)
target_compile_options(qimgvr PRIVATE -Wall -Wextra)
pkg_check_modules(MPV_HEADERS REQUIRED mpv)
find_library(DE_XR_MPV_LIBRARY NAMES mpv PATHS "${CMAKE_CURRENT_SOURCE_DIR}/../../mpvr/build" NO_DEFAULT_PATH)
if(NOT DE_XR_MPV_LIBRARY)
  message(FATAL_ERROR "Build the BDF-enabled mpvr libmpv first or set DE_XR_MPV_LIBRARY")
endif()
target_include_directories(qimgvr PRIVATE ${MPV_HEADERS_INCLUDE_DIRS})
target_link_libraries(qimgvr PRIVATE qimgvr-model qimgvr-gltf qimgvr-image fltk::gl-shared OpenGL::GL PkgConfig::MEDIA fltk::fltk-shared fltk::images-shared PkgConfig::FONTCONFIG ${DE_XR_MPV_LIBRARY})
set(DE_XR_BITMAP_ASS "" CACHE FILEPATH "Patched libass with native BDF support")
if(NOT DE_XR_BITMAP_ASS)
  find_library(DE_XR_DISCOVERED_ASS NAMES ass PATHS
    "${CMAKE_CURRENT_SOURCE_DIR}/../../fltk-experiment/libass/build-bitmap/libass"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../libass/build-bitmap/libass"
    /opt/fltk-vr/lib/bitmap NO_DEFAULT_PATH)
  if(NOT DE_XR_DISCOVERED_ASS)
    message(FATAL_ERROR "Set DE_XR_BITMAP_ASS to the BDF-enabled libass build")
  endif()
  set(DE_XR_BITMAP_ASS "${DE_XR_DISCOVERED_ASS}")
endif()
get_filename_component(DE_XR_BITMAP_ASS_DIR "${DE_XR_BITMAP_ASS}" DIRECTORY)
target_link_options(qimgvr PRIVATE -Wl,--disable-new-dtags)
get_target_property(DE_XR_FLTK_LIBRARY fltk::fltk-shared IMPORTED_LOCATION_RELEASE)
get_filename_component(DE_XR_FLTK_LIBRARY_DIR "${DE_XR_FLTK_LIBRARY}" DIRECTORY)
set_target_properties(qimgvr PROPERTIES BUILD_RPATH "${DE_XR_FLTK_LIBRARY_DIR};${DE_XR_BITMAP_ASS_DIR};${CMAKE_CURRENT_SOURCE_DIR}/../../mpvr/build" INSTALL_RPATH "$ORIGIN/../lib;$ORIGIN/../lib/bitmap;${DE_XR_BITMAP_ASS_DIR}")
include(GNUInstallDirs)
install(TARGETS qimgvr RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES qimgvr.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)

include(desktop-menu.cmake)
de_xr_desktop_menu(qimgvr)
include(CTest)

  target_link_libraries(qimgvr-image-test PRIVATE qimgvr-image)
  add_test(NAME native-images COMMAND qimgvr-image-test)
endif()

  target_compile_features(qimgvr-canvas-test PRIVATE cxx_std_17)
  target_include_directories(qimgvr-canvas-test PRIVATE ${MPV_HEADERS_INCLUDE_DIRS})
  target_link_libraries(qimgvr-canvas-test PRIVATE qimgvr-image fltk::gl-shared OpenGL::GL PkgConfig::MEDIA fltk::fltk-shared fltk::images-shared PkgConfig::FONTCONFIG ${DE_XR_MPV_LIBRARY})
endif()

 DESTINATION ${CMAKE_INSTALL_DATADIR}/qimgvr/licenses)
