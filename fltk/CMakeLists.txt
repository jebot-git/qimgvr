find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
pkg_check_modules(MEDIA REQUIRED IMPORTED_TARGET mpv exiv2 egl gio-2.0)
find_package(OpenCV REQUIRED core imgcodecs imgproc)
find_package(OpenGL REQUIRED)
add_library(qimgvr-image STATIC image.cpp)
target_compile_features(qimgvr-image PUBLIC cxx_std_17)
target_link_libraries(qimgvr-image PUBLIC fltk::images-shared ${OpenCV_LIBS} PkgConfig::MEDIA)
target_include_directories(qimgvr-image PUBLIC ${OpenCV_INCLUDE_DIRS})
add_executable(qimgvr main.cpp canvas.cpp stereo.cpp)
target_compile_features(qimgvr PRIVATE cxx_std_17)
target_compile_options(qimgvr PRIVATE -Wall -Wextra)
target_link_libraries(qimgvr PRIVATE qimgvr-image fltk::gl-shared OpenGL::GL PkgConfig::MEDIA fltk::fltk-shared fltk::images-shared PkgConfig::FONTCONFIG)
include(GNUInstallDirs)
install(TARGETS qimgvr RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

include(desktop-menu.cmake)
de_xr_desktop_menu(qimgvr)
include(CTest)
if(BUILD_TESTING)
  add_executable(qimgvr-image-test image-test.cpp)
  target_link_libraries(qimgvr-image-test PRIVATE qimgvr-image)
  add_test(NAME native-images COMMAND qimgvr-image-test)
endif()
